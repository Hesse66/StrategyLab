trimLiveArt() =>

isDispBar(float o, float c, float h, float l, float impulse) =>
    body = math.abs(c - o)
    rng  = h - l
    ratio = rng > 0 ? body / rng : 0.0
    ratio >= 0.55 and body >= impulse

lastPivotKind(array<Pivot> arr, int k) =>
    Pivot found = na
    sz = array.size(arr)
    if sz > 0
        for i = sz - 1 to 0
            p = array.get(arr, i)
            if p.kind == k
                found := p
                break
    found

findOrigin(int dir) =>
    // Scan a 30-bar structural window backward from the break.
    // Pick the most-extreme candle that passes three independent quality filters:
    //   (1) opposite-color candle
    //   (2) not abnormally wide (range < ATR * 2.5) — volatility filter
    //   (3) decent body-to-range ratio (>= 0.25) — quality filter, unique here
    //   (4) above-average volume (optional, via obVolReq toggle)
    int    oBar = -1
    float  oTop = na
    float  oBtm = na
    for i = 1 to 30
        o_ = open[i]
        c_ = close[i]
        h_ = high[i]
        l_ = low[i]
        v_ = volume[i]
        va_ = volAvg[i]
        rng = h_ - l_
        body = math.abs(c_ - o_)
        bodyRatio = rng > 0 ? body / rng : 0.0
        volOk       = obVolReq ? (nz(v_) >= nz(va_) * obVolMult) : true
        notVolatile = rng < atrFast * 2.5
        qualityOk   = bodyRatio >= 0.25
        opposite    = dir == 1 ? c_ < o_ : c_ > o_
        if opposite and notVolatile and volOk and qualityOk
            if dir == 1
                // Bull OB: track lowest low among qualifying candles
                if na(oBtm) or l_ < oBtm
                    oBar := bar_index - i
                    oTop := h_
                    oBtm := l_
            else
                // Bear OB: track highest high among qualifying candles
                if na(oTop) or h_ > oTop
                    oBar := bar_index - i
                    oTop := h_
                    oBtm := l_
    [oBar, oTop, oBtm]

computeObScore(float top, float bottom, int dir) =>
    sz       = (top - bottom) / math.max(atrFast, 0.0001)
    sizeScr  = math.min(sz / 2.0, 1.0) * 30.0
    dispBody = math.abs(close - open)
    dispScr  = math.min(dispBody / math.max(atrFast * 1.5, 0.0001), 1.0) * 30.0
    volScr   = volume > volAvg ? 20.0 : 10.0
    float confScr = 0.0
    fvgs = dir == 1 ? bullFVGs : bearFVGs
    for f in fvgs
        if f.top >= bottom and f.bottom <= top
            confScr := 20.0
    sizeScr + dispScr + volScr + confScr

pushOB(int dir) =>
    [oBar, oTop, oBtm] = findOrigin(dir)
    if oBar > 0 and not na(oTop) and not na(oBtm)
        score = computeObScore(oTop, oBtm, dir)
        if score >= minScore
            col = dir == 1 ? bullObCol : bearObCol
            bx = box.new(oBar, oTop, bar_index + 5, oBtm,
                 bgcolor      = col,
                 border_color = color.new(col, 40),
                 extend       = extend.right)
            label tag = na
            if showScore
                tag := label.new(bar_index + 5, (oTop + oBtm) / 2, str.tostring(math.round(score)),
                     color     = color.new(col, 60),
                     textcolor = color.white,
                     style     = label.style_label_left,
                     size      = size.tiny)
            z = Zone.new(oBar, oTop, oBtm, dir, bar_index, 0, score, false, bx, tag)
            arr = dir == 1 ? bullOBs : bearOBs
            array.push(arr, z)
            while array.size(arr) > obMax
                old = array.shift(arr)
                box.delete(old.bx)
                if not na(old.tag)
                    label.delete(old.tag)

// OB mitigation + extension
mitigateOBs(array<Zone> arr, int dir) =>
    sz = array.size(arr)
    if sz > 0
        for i = sz - 1 to 0
            z = array.get(arr, i)
            bool dead = false
            if dir == 1
                dead := obMitig == "Wick" ? low < z.bottom : close < z.bottom
            else
                dead := obMitig == "Wick" ? high > z.top    : close > z.top
            if dead
                box.delete(z.bx)
                if not na(z.tag)
                    label.delete(z.tag)
                array.remove(arr, i)
            if not dead
                box.set_right(z.bx, bar_index + 5)
                if not na(z.tag)
                    label.set_x(z.tag, bar_index + 5)
                    label.set_y(z.tag, (z.top + z.bottom) / 2)
    0

// A sweep requires ALL of:
//   (1) wick pokes through a recent MINOR pivot (real liquidity pool,

//   (2) close returns back inside the range
//   (3) wick length >= 0.8 × ATR (substantial rejection)
//   (4) wick >= 35% of the candle's total range (dominant wick)
//   (5) minimum 8 bars since the last same-direction sweep (cooldown)

bool bearSweepSig = false

if showSweep
    topWick = high - math.max(open, close)
    botWick = math.min(open, close) - low
    rng_    = high - low

    // Find the nearest minor pivot highs/lows to use as liquidity references
    float refHi = na
    float refLo = na
    sz = array.size(minorPivots)
    if sz > 0
        maxScan = math.max(sz - 12, 0)
        for i = sz - 1 to maxScan
            p = array.get(minorPivots, i)
            if na(refHi) and p.kind == 1 and bar_index - p.bar > 2
                refHi := p.price
            if na(refLo) and p.kind == -1 and bar_index - p.bar > 2
                refLo := p.price
            if not na(refHi) and not na(refLo)
                break

    bearSweepSig := not na(refHi) 
         and high > refHi 
         and close < refHi 
         and topWick >= atrFast * 0.8 
         and topWick / math.max(rng_, 0.0001) >= 0.35 
         and bar_index - lastBearSweep > 8

    bullSweepSig := not na(refLo) 
         and low < refLo 
         and close > refLo 
         and botWick >= atrFast * 0.8 
         and botWick / math.max(rng_, 0.0001) >= 0.35 
         and bar_index - lastBullSweep > 8

    if bearSweepSig
        lastBearSweep := bar_index
        label.new(bar_index, high + atrFast * 0.2, "✕",
             color     = color.new(bearC, 0),
             textcolor = color.white,
             style     = label.style_label_down,
             size      = size.small)
    if bullSweepSig
        lastBullSweep := bar_index
        label.new(bar_index, low - atrFast * 0.2, "✕",
             color     = color.new(bullC, 0),
             textcolor = color.white,
             style     = label.style_label_up,
             size      = size.small)

// Running AVWAP since last major structure event
if bar_index >= major.anchorBar
    safeVol = nz(volume, 1) // without volume, = 1
    hlc3v = (high + low + close) / 3.0 * safeVol
    major.anchorVwapNum := major.anchorVwapNum + hlc3v
    major.anchorVwapDen := major.anchorVwapDen + safeVol

avwap = major.anchorVwapDen > 0 ? major.anchorVwapNum / major.anchorVwapDen : close

var line   premL = na
var line   eqL   = na
var line   discL = na
var label  premB = na
var label  eqB   = na
var label  discB = na

if barstate.islast and showPD
    lookBack = math.min(bar_index - major.anchorBar, 500)
    hh = high
    ll = low
    if lookBack > 0
        for i = 0 to lookBack
            hh := math.max(hh, high[i])
            ll := math.min(ll, low[i])
    equilibrium = pdMethod == "AVWAP Equilibrium" ? avwap : math.avg(hh, ll)

    line.delete(premL)
    line.delete(eqL)
    line.delete(discL)
    label.delete(premB)
    label.delete(eqB)
    label.delete(discB)

    premL := line.new(major.anchorBar, hh, bar_index + 20, hh,
         color = bearC, width = 1)
    eqL   := line.new(major.anchorBar, equilibrium, bar_index + 20, equilibrium,
         color = color.new(color.gray, 30), style = line.style_dashed)
    discL := line.new(major.anchorBar, ll, bar_index + 20, ll,
         color = bullC, width = 1)
    premB := label.new(bar_index + 20, hh, "Premium",
         color     = color.new(bearC, 100),
         textcolor = bearC,
         style     = label.style_label_left,
         size      = size.small)
    eqB := label.new(bar_index + 20, equilibrium,
         pdMethod == "AVWAP Equilibrium" ? "AVWAP EQ" : "EQ 50%",
         color     = color.new(color.gray, 100),
         textcolor = color.gray,
         style     = label.style_label_left,
         size      = size.tiny)
    discB := label.new(bar_index + 20, ll, "Discount",

         style     = label.style_label_left,
         size      = size.small)
